local CorrectKey = "MiranDaHubKeyHdiehBsiwibesoHdieoejkdoebejb"

-- Hàm kiểm tra Key đã lưu trong 24 giờ hay chưa
local function CheckSavedKey()
    if isfile and readfile and isfile(FileName) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(FileName))
        end)
        if success and data and data.key == CorrectKey then
            -- Kiểm tra thời gian (24 giờ = 86400 giây)
            if os.time() - data.time < 86400 then
                return true
            end
        end
    end
    return false
end

-- Hàm lưu Key lại kèm theo timestamp hiện tại
local function SaveKey()
    if writefile then
        local data = {
            key = CorrectKey,
            time = os.time()
        }
        pcall(function()
            writefile(FileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Nếu Key đã được lưu và hợp lệ, bỏ qua bảng Key và load thẳng script chính
if CheckSavedKey() then
    loadstring(game:HttpGet("https://raw.githubusercontent.com/miirandahub/loader/refs/heads/main/mirandaafk.lua"))()
    return
end

-- Ultra-Smooth Sine-Wave Shake Animation
local function SmoothShake()
    local startTime = tick()
    local duration = 0.35
    local intensity = 10
    local frequency = 35
    
    local connection
    connection = game:GetService("RunService").RenderStepped:Connect(function()
        local elapsed = tick() - startTime
        if elapsed < duration then
            local progress = 1 - (elapsed / duration)
            local offsetX = math.sin(elapsed * frequency) * intensity * progress
            MainFrame.Position = UDim2.new(0.5, offsetX, 0.5, 0)
        else
            MainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
            connection:Disconnect()
        end
    end)
end

-- Smooth Text Clearing Effect
local function SmoothClearText()
    local currentText = TextBox.Text
    while #currentText > 0 do
        currentText = string.sub(currentText, 1, #currentText - 1)
        TextBox.Text = currentText
        task.wait(0.02)
    end
end
